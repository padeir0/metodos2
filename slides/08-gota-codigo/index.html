double initial(double x, double y) {
  const double x0 = 0.35;
  const double y0 = 0.50;
  const double sigma = 0.06;
  double dx = x - x0;
  double dy = y - y0;
  return exp(-(dx*dx + dy*dy) / (2*sigma*sigma));
}

double noWall(double x, double y) { (void)x; (void)y; return 0; }
