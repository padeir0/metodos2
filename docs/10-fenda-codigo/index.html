Exemplo 2: fenda dupla


double initial(double x, double y) {
  double dx = x - SOURCE_X;
  double dy = y - SOURCE_Y;
  return exp(-(dx*dx + dy*dy)
             / (2*SIGMA*SIGMA));
}

// uma fenda de largura SLIT_WIDTH
// centrada em `center`
static int inSlit(double y, double center) {
  return fabs(y - center) < SLIT_WIDTH / 2;
}

// obstáculo: parede vertical em x = WALL_X
// com duas fendas simétricas em torno de y = 0.5
double wall(double x, double y) {
  if (fabs(x - WALL_X) >= WALL_THICKNESS / 2) {
    return 0;
  }
  int open =
      inSlit(y, 0.5 - SLIT_SEPARATION/2)
      || inSlit(y, 0.5 + SLIT_SEPARATION/2);
  return !open;
}
